rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper functions
    function isLoggedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isLoggedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Users collectie - alleen ingelogde users kunnen hun eigen document lezen
    match /users/{userId} {
      allow read: if isLoggedIn() && request.auth.uid == userId;
      allow write: if isLoggedIn() && request.auth.uid == userId;
      allow create: if isLoggedIn();
    }

    // Race Teams per user - subcollection teams
    match /users/{userId}/teams/{raceId} {
      allow read: if isLoggedIn() && request.auth.uid == userId;
      allow write: if isLoggedIn() && request.auth.uid == userId;
      allow create: if isLoggedIn() && request.auth.uid == userId;
      allow delete: if isLoggedIn() && request.auth.uid == userId;
    }

    // Teams (user's complete team/budget team) - users kunnen hun eigen team lezen/schrijven
    match /teams/{userId} {
      allow read: if isLoggedIn() && request.auth.uid == userId;
      allow write: if isLoggedIn() && request.auth.uid == userId;
      allow create: if isLoggedIn() && request.auth.uid == userId;
      allow delete: if isLoggedIn() && request.auth.uid == userId;
    }

    // Riders - iedereen kan lezen, alleen admins kunnen schrijven
    match /riders/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Cycling Teams - iedereen kan lezen, alleen admins kunnen schrijven
    match /cyclingTeams/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Races - iedereen kan lezen, alleen admins kunnen schrijven
    match /races/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Race Categories - iedereen kan lezen, alleen admins kunnen schrijven
    match /raceCategories/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Points by Category - iedereen kan lezen, alleen admins kunnen schrijven
    match /pointsPerCategory/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Results - iedereen kan lezen, alleen admins kunnen schrijven
    match /results/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Race Participants - iedereen kan lezen, alleen admins kunnen schrijven
    match /raceParticipants/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }
  }
}
