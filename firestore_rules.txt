rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Helper functions
    function isLoggedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isLoggedIn()
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // Users collectie - create allowed for new users, read all users for logged in, write only self
    match /users/{userId} {
      allow create: if isLoggedIn();
      allow read: if isLoggedIn(); // Allow all logged-in users to read any user (needed for Rankings)
      allow write: if isLoggedIn() && request.auth.uid == userId;
    }

    // Race Teams per user - subcollection teams
    // Allow all logged-in users to read (needed for Rankings/TeamDetails display)
    match /users/{userId}/teams/{raceId} {
      allow read: if isLoggedIn();
      allow write: if isLoggedIn() && (request.auth.uid == userId || isAdmin());
      allow create: if isLoggedIn() && (request.auth.uid == userId || isAdmin());
      allow delete: if isLoggedIn() && (request.auth.uid == userId || isAdmin());
    }

    // Teams (user's complete team/budget team) - read allowed for all logged in users
    match /teams/{userId} {
      allow read: if isLoggedIn();
      allow write: if isLoggedIn() 
        && (request.auth.uid == userId || isAdmin());
      allow create: if isLoggedIn() 
        && (request.auth.uid == userId || isAdmin());
      allow delete: if isLoggedIn() && (request.auth.uid == userId || isAdmin());
    }

    // Riders - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /riders/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Cycling Teams - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /cyclingTeams/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Races - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /races/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Race Categories - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /raceCategories/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Points by Category - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /pointsPerCategory/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Results - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /results/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }

    // Race Participants - iedereen ingelogd kan lezen, alleen admins kunnen schrijven
    match /raceParticipants/{document=**} {
      allow read: if isLoggedIn();
      allow write: if isAdmin();
    }
  }
}
